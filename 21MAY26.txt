-- GROUP BY

-- Count total workers
select count(worker_id) from worker;

-- Count workers department wise
select department, count(department)
from worker
group by department
order by department asc;

-- Departments having 3 or more employees
select department, count(department)
from worker
group by department
having count(department) >= 3
order by department asc;



-- ORDER BY

-- Admin employees salary in descending order
select first_name, department, salary
from worker
where department='admin'
order by salary desc;

-- Sort workers by salary ascending
select *
from worker
order by salary asc;

-- Sort workers by first name descending
select *
from worker
order by first_name desc;



-- HAVING

-- Departments having less than 3 employees
select department, count(department)
from worker
group by department
having count(department) < 3;

-- Departments having more than 2 employees
select department, count(department)
from worker
group by department
having count(department) > 2;



-- LIMIT AND OFFSET

-- Fetch 2 records after skipping first 3 rows
select *
from worker
limit 2 offset 3;



-- SUBQUERY

-- Create toppers table
create table toppers(
    id int
);

-- Insert values into toppers
insert into toppers values(2),(5);

-- View toppers table
select * from toppers;

-- Fetch workers whose id exists in toppers table
select worker_id, first_name, department
from worker
where worker_id in (select id from toppers);

-- Another subquery example
select *
from worker
where salary = (
    select max(salary)
    from worker
);

-- Second highest salary using subquery
select max(salary) as second_highest_salary
from worker
where salary < (
    select max(salary)
    from worker
);
